> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Add all 5 endpoints of the "Dogma" ESI tag as a new `esi/dogma.go` facade and a new `mcp/dogma.go` tool file, registered on the shared MCP server.

**Architecture:** One new `esi/` facade file wrapping the generated `DogmaAPI`, reusing the existing `describeError` helper. One new `mcp/` tool file with 5 public tools (`esi.NewPublicClient`, no existence check, no `characterId` input) — this tag is fully public reference data, unlike the corp-scoped batches immediately preceding it.

**Tech Stack:** Go 1.26.1, `github.com/fnt-eve/goesi-openapi` v0.0.37, `github.com/modelcontextprotocol/go-sdk`.

## Global Constraints

- This plan builds directly on the shared PR branch `worktree-esi-access-list-activities` (current HEAD `2bd6eab`) — no new branch, no new PR.
- Current tool count baseline: **83**. After this plan: **88**.
- Every `esi/` method's `.Execute()` call must route its error and `*http.Response` through the existing `describeError` (`esi/error.go`) — do not redeclare it.
- All 5 tools are **public** — use `esi.NewPublicClient(userAgent string) *Client`, no `characterExists` check, no `characterId` input field.
- **`DogmaDynamicItemDetail`'s parameter order**: the generated function `GetDogmaDynamicItemsTypeIdItemId(ctx, itemId, typeId)` takes `itemId` before `typeId` (reversed from its own URL path `/dogma/dynamic/items/{type_id}/{item_id}/`) — verified from source. neat's own facade method keeps `(typeID, itemID)` order (matching the URL, and matching the tool's own `TypeID`-then-`ItemID` input field order) and internally swaps when calling the generated function, exactly like the Contracts batch's `CorporationContractBids` reversed-argument precedent.
- No pagination on any of these 5 endpoints.
- All code below has already been verified to compile against the real installed `github.com/fnt-eve/goesi-openapi@v0.0.37` module in an isolated sandbox (`go build`/`go vet`/`gofmt -l` all clean). Implementers should still run these themselves as part of each task.

// Package esi's dogma.go covers all 5 of "Dogma"'s endpoints -- Dogma is EVE's live
// attribute/effect reference data (item stats, module effects, mutated/"abyssal" item detail),
// all public, no character auth required.
package esi

import (
	"context"
	"fmt"

	esiapi "github.com/fnt-eve/goesi-openapi/esi"
)

// DogmaAttributes returns every dogma attribute ID currently defined.
func (c *Client) DogmaAttributes(ctx context.Context) ([]int64, error) {
	result, resp, err := c.api.DogmaAPI.GetDogmaAttributes(ctx).Execute()
	if err != nil {
		return nil, fmt.Errorf("get dogma attributes: %w", describeError(err, resp))
	}
	return result, nil
}

// DogmaAttributeDetail returns detail for one dogma attribute.
func (c *Client) DogmaAttributeDetail(ctx context.Context, attributeID int64) (*esiapi.DogmaAttributesAttributeIdGet, error) {
	result, resp, err := c.api.DogmaAPI.GetDogmaAttributesAttributeId(ctx, attributeID).Execute()
	if err != nil {
		return nil, fmt.Errorf("get dogma attribute detail: %w", describeError(err, resp))
	}
	return result, nil
}

// DogmaDynamicItemDetail returns detail for one mutated ("abyssal") item, identified by its
// source type and item ID. Note: the generated function's own argument order is
// (itemId, typeId) -- reversed from its URL path -- this facade normalizes to (typeID, itemID)
// to match the URL and this method's own signature, swapping internally.
func (c *Client) DogmaDynamicItemDetail(ctx context.Context, typeID int64, itemID int64) (*esiapi.DogmaDynamicItemsTypeIdItemIdGet, error) {
	result, resp, err := c.api.DogmaAPI.GetDogmaDynamicItemsTypeIdItemId(ctx, itemID, typeID).Execute()
	if err != nil {
		return nil, fmt.Errorf("get dogma dynamic item detail: %w", describeError(err, resp))
	}
	return result, nil
}

// DogmaEffects returns every dogma effect ID currently defined.
func (c *Client) DogmaEffects(ctx context.Context) ([]int64, error) {
	result, resp, err := c.api.DogmaAPI.GetDogmaEffects(ctx).Execute()
	if err != nil {
		return nil, fmt.Errorf("get dogma effects: %w", describeError(err, resp))
	}
	return result, nil
}

// DogmaEffectDetail returns detail for one dogma effect.
func (c *Client) DogmaEffectDetail(ctx context.Context, effectID int64) (*esiapi.DogmaEffectsEffectIdGet, error) {
	result, resp, err := c.api.DogmaAPI.GetDogmaEffectsEffectId(ctx, effectID).Execute()
	if err != nil {
		return nil, fmt.Errorf("get dogma effect detail: %w", describeError(err, resp))
	}
	return result, nil
}

package mcp

import (
	"context"
	"fmt"

	mcpsdk "github.com/modelcontextprotocol/go-sdk/mcp"

	"github.com/rusher2004/new-eden-agent-toolkit/esi"
	"github.com/rusher2004/new-eden-agent-toolkit/sso"

// --- list_dogma_attributes (public, no input) ---

var listDogmaAttributes = func(ctx context.Context, userAgent string) ([]int64, error) {
	client := esi.NewPublicClient(userAgent)
	return client.DogmaAttributes(ctx)
}

type ListDogmaAttributesInput struct{}
type ListDogmaAttributesOutput struct {
	AttributeIDs []int64 `json:"attributeIds"`
}

func listDogmaAttributesHandler(version string) mcpsdk.ToolHandlerFor[ListDogmaAttributesInput, ListDogmaAttributesOutput] {
	return func(ctx context.Context, _ *mcpsdk.CallToolRequest, in ListDogmaAttributesInput) (*mcpsdk.CallToolResult, ListDogmaAttributesOutput, error) {
		ids, err := listDogmaAttributes(ctx, sso.UserAgent(version))
		if err != nil {
			return nil, ListDogmaAttributesOutput{}, fmt.Errorf("list dogma attributes: %w", err)
		}
		return nil, ListDogmaAttributesOutput{AttributeIDs: ids}, nil
	}
}

// --- get_dogma_attribute (public) ---

type GetDogmaAttributeOutput struct {
	AttributeID  int64    `json:"attributeId"`
	DefaultValue *float64 `json:"defaultValue,omitempty"`
	Description  *string  `json:"description,omitempty"`
	DisplayName  *string  `json:"displayName,omitempty"`
	HighIsGood   *bool    `json:"highIsGood,omitempty"`
	IconID       *int64   `json:"iconId,omitempty"`
	Name         *string  `json:"name,omitempty"`
	Published    *bool    `json:"published,omitempty"`
	Stackable    *bool    `json:"stackable,omitempty"`
	UnitID       *int64   `json:"unitId,omitempty"`
}

var getDogmaAttribute = func(ctx context.Context, userAgent string, attributeID int64) (GetDogmaAttributeOutput, error) {
	client := esi.NewPublicClient(userAgent)
	attr, err := client.DogmaAttributeDetail(ctx, attributeID)
	if err != nil {
		return GetDogmaAttributeOutput{}, err
	}
	return GetDogmaAttributeOutput{
		AttributeID: attr.AttributeId, DefaultValue: attr.DefaultValue, Description: attr.Description,
		DisplayName: attr.DisplayName, HighIsGood: attr.HighIsGood, IconID: attr.IconId,
		Name: attr.Name, Published: attr.Published, Stackable: attr.Stackable, UnitID: attr.UnitId,
	}, nil
}

type GetDogmaAttributeInput struct {
	AttributeID int64 `json:"attributeId" jsonschema:"the dogma attribute ID to look up"`
}

func getDogmaAttributeHandler(version string) mcpsdk.ToolHandlerFor[GetDogmaAttributeInput, GetDogmaAttributeOutput] {
	return func(ctx context.Context, _ *mcpsdk.CallToolRequest, in GetDogmaAttributeInput) (*mcpsdk.CallToolResult, GetDogmaAttributeOutput, error) {
		out, err := getDogmaAttribute(ctx, sso.UserAgent(version), in.AttributeID)

			return nil, GetDogmaAttributeOutput{}, fmt.Errorf("get dogma attribute: %w", err)

// --- get_dogma_dynamic_item (public) ---

type DynamicItemAttribute struct {
	AttributeID int64   `json:"attributeId"`
	Value       float64 `json:"value"`
}
type DynamicItemEffect struct {
	EffectID  int64 `json:"effectId"`
	IsDefault bool  `json:"isDefault"`
}

type GetDogmaDynamicItemOutput struct {
	TypeID          int64                  `json:"typeId"`
	ItemID          int64                  `json:"itemId"`
	CreatedBy       int64                  `json:"createdBy"`
	MutatorTypeID   int64                  `json:"mutatorTypeId"`
	SourceTypeID    int64                  `json:"sourceTypeId"`
	DogmaAttributes []DynamicItemAttribute `json:"dogmaAttributes"`
	DogmaEffects    []DynamicItemEffect    `json:"dogmaEffects"`
}

var getDogmaDynamicItem = func(ctx context.Context, userAgent string, typeID, itemID int64) (GetDogmaDynamicItemOutput, error) {
	client := esi.NewPublicClient(userAgent)
	item, err := client.DogmaDynamicItemDetail(ctx, typeID, itemID)
	if err != nil {
		return GetDogmaDynamicItemOutput{}, err
	}
	attrs := make([]DynamicItemAttribute, len(item.DogmaAttributes))
	for i, a := range item.DogmaAttributes {
		attrs[i] = DynamicItemAttribute{AttributeID: a.AttributeId, Value: a.Value}
	}
	effects := make([]DynamicItemEffect, len(item.DogmaEffects))
	for i, e := range item.DogmaEffects {
		effects[i] = DynamicItemEffect{EffectID: e.EffectId, IsDefault: e.IsDefault}
	}
	return GetDogmaDynamicItemOutput{
		TypeID: typeID, ItemID: itemID, CreatedBy: item.CreatedBy, MutatorTypeID: item.MutatorTypeId,
		SourceTypeID: item.SourceTypeId, DogmaAttributes: attrs, DogmaEffects: effects,
	}, nil
}

type GetDogmaDynamicItemInput struct {
	TypeID int64 `json:"typeId" jsonschema:"the mutated item's type ID"`
	ItemID int64 `json:"itemId" jsonschema:"the mutated item's item ID"`
}
