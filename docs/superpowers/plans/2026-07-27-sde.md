# `sde/` Package Infrastructure Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the `sde/` package — a `bbolt`-backed local cache of CCP's Static Data Export, covering categories/groups/types, syncing lazily and atomically from EVE Ref, with no dependency on any other package and no consumers wired up yet.

**Architecture:** `sde.Store` wraps a `*bbolt.DB`. Its first lookup call per process lifetime (`sync.Once`) checks EVE Ref's live `index.json` `etag` against a version cached in a `_meta` bucket inside the same `sde.db` file; a stale or missing version triggers an in-memory zip download, streamed JSONL parse, and single atomic `bbolt` transaction rewriting `categories`/`groups`/`types` plus the new version together. A failed sync is swallowed at the accessor boundary — lookups fall back to whatever's already cached rather than erroring, since SDE data is enrichment, not a hard requirement.

**Tech Stack:** Go 1.26.1, `go.etcd.io/bbolt` v1.5.0 (new dependency), standard library only otherwise (`archive/zip`, `bufio`, `net/http`).

```go
// Package sde caches CCP's Static Data Export locally in a bbolt database,
// so tools can resolve ESI's bare IDs into names without hitting ESI's own
// (much more limited, rate-limited) name-resolution endpoints.
package sde

import (
	"fmt"
	"os"
	"path/filepath"
	"sync"

	"go.etcd.io/bbolt"
)

const defaultIndexURL = "https://data.everef.net/ccp/sde/index.json"

// Store is a local cache of CCP's Static Data Export, backed by bbolt.
// Opening a Store does not sync it -- the first lookup call does that
// lazily, at most once per Store's lifetime.
type Store struct {
	db       *bbolt.DB
	indexURL string

	syncOnce sync.Once
	syncErr  error
}

// Open opens (creating if necessary) the bbolt database at path.
func Open(path string) (*Store, error) {
	if err := os.MkdirAll(filepath.Dir(path), 0o700); err != nil {
		return nil, fmt.Errorf("create sde cache directory: %w", err)
	}
	db, err := bbolt.Open(path, 0o600, nil)
	if err != nil {
		return nil, fmt.Errorf("open sde cache: %w", err)
	}
	return &Store{db: db, indexURL: defaultIndexURL}, nil
}

// DefaultPath returns ~/.neat/sde/sde.db.
func DefaultPath() (string, error) {
	home, err := os.UserHomeDir()
	if err != nil {
		return "", fmt.Errorf("resolve home directory: %w", err)
	}
	return filepath.Join(home, ".neat", "sde", "sde.db"), nil
}

// Close closes the underlying bbolt database.
func (s *Store) Close() error {
	return s.db.Close()
}
```

```go
func buildZip(t *testing.T, files map[string][]string) []byte {
	t.Helper()
	var buf bytes.Buffer
	zw := zip.NewWriter(&buf)
	for name, lines := range files {
		w, err := zw.Create(name)
		if err != nil {
			t.Fatalf("create zip entry %s: %v", name, err)

		}
```

```go
// ExportedOpenWithIndexURL opens a Store exactly like Open, but overrides
// the SDE index URL -- for external tests that need to point sync at an
// httptest.Server instead of the real EVE Ref endpoint.
func ExportedOpenWithIndexURL(path, indexURL string) (*Store, error) {
	store, err := Open(path)
	if err != nil {
		return nil, err
	}
	store.indexURL = indexURL
	return store, nil
}

// ExportedSyncForTest runs (or reuses the cached result of) this Store's
// lazy sync -- for external tests that need to drive the sync path
// directly, before Category/Group/Type exist to trigger it implicitly.
func (s *Store) ExportedSyncForTest(ctx context.Context, userAgent string) error {
	return s.ensureSynced(ctx, userAgent)
}

// ExportedGetForTest returns the raw cached bytes for id in bucket, or nil
// if there's no cached record -- for external tests verifying sync results
// directly, without going through a resource-specific accessor.
func (s *Store) ExportedGetForTest(bucket string, id int64) ([]byte, error) {
	var v []byte
	err := s.db.View(func(tx *bbolt.Tx) error {
		b := tx.Bucket([]byte(bucket))
		if b == nil {
			return nil
		}
		if raw := b.Get(int64Key(id)); raw != nil {
			v = append([]byte(nil), raw...)
		}
		return nil
	})
	return v, err
}
```

```go
package sde

import (
	"archive/zip"
	"bufio"
	"bytes"
	"context"
	"encoding/binary"
	"encoding/json"
	"fmt"
	"io"
	"net/http"

	"go.etcd.io/bbolt"
)

const sdeZipName = "eve-online-static-data-latest-jsonl.zip"

// syncedFiles maps each JSONL file inside the SDE archive to the bbolt
// bucket its records are stored under.
var syncedFiles = map[string]string{
	"categories.jsonl": "categories",
	"groups.jsonl":     "groups",
	"types.jsonl":      "types",
}

type sdeIndex struct {
	Files []sdeIndexFile `json:"files"`
}

type sdeIndexFile struct {
	Name string `json:"name"`
	URL  string `json:"url"`
	Etag string `json:"etag"`
}

// ensureSynced checks the store's cached SDE version against EVE Ref's live
// index exactly once per Store lifetime (via s.syncOnce), downloading and
// reloading categories/groups/types if it's stale or missing. A failed sync
// is cached too -- lookups fall back to whatever's already in bbolt (empty,
// on a never-synced install) rather than retrying every call. SDE data is
// enrichment, not a hard requirement for any ESI tool to function.
func (s *Store) ensureSynced(ctx context.Context, userAgent string) error {
	s.syncOnce.Do(func() {
		s.syncErr = s.sync(ctx, userAgent)
	})
	return s.syncErr
}

func (s *Store) sync(ctx context.Context, userAgent string) error {
	remoteEtag, zipURL, err := latestZipInfo(ctx, s.indexURL, userAgent)
	if err != nil {
		return fmt.Errorf("fetch sde index: %w", err)
	}

	currentVersion, err := s.version()
	if err != nil {
		return fmt.Errorf("read cached sde version: %w", err)
	}
	if currentVersion == remoteEtag {
		return nil
	}

	body, err := fetchBytes(ctx, zipURL, userAgent)
	if err != nil {
		return fmt.Errorf("download sde archive: %w", err)
	}
	zr, err := zip.NewReader(bytes.NewReader(body), int64(len(body)))
	if err != nil {
		return fmt.Errorf("open sde archive: %w", err)
	}

	return s.db.Update(func(tx *bbolt.Tx) error {
		for file, bucket := range syncedFiles {
			if err := loadResource(tx, zr, file, bucket); err != nil {
				return fmt.Errorf("load %s: %w", file, err)
			}
		}
		meta, err := tx.CreateBucketIfNotExists([]byte("_meta"))
		if err != nil {
			return fmt.Errorf("create _meta bucket: %w", err)
		}
		return meta.Put([]byte("version"), []byte(remoteEtag))
	})
}

// version returns the SDE version currently cached in bbolt's _meta bucket,
// or "" if the store has never been synced.
func (s *Store) version() (string, error) {
	var version string
	err := s.db.View(func(tx *bbolt.Tx) error {
		meta := tx.Bucket([]byte("_meta"))
		if meta == nil {
			return nil
		}
		version = string(meta.Get([]byte("version")))
		return nil
	})
	return version, err
}

// loadResource clears and rewrites bucket from file's JSONL lines within
// the zip, keyed by each record's _key field.
func loadResource(tx *bbolt.Tx, zr *zip.Reader, file, bucket string) error {
	f, err := zr.Open(file)
	if err != nil {
		return fmt.Errorf("open zip entry: %w", err)
	}
	defer f.Close()

	if err := tx.DeleteBucket([]byte(bucket)); err != nil && err != bbolt.ErrBucketNotFound {
		return fmt.Errorf("clear bucket: %w", err)
	}
	b, err := tx.CreateBucket([]byte(bucket))
	if err != nil {
		return fmt.Errorf("create bucket: %w", err)
	}

	scanner := bufio.NewScanner(f)
	scanner.Buffer(make([]byte, 0, 64*1024), 4*1024*1024) // types.jsonl has long lines
	for scanner.Scan() {
		line := scanner.Bytes()
		var keyed struct {
			Key int64 `json:"_key"`
		}
		if err := json.Unmarshal(line, &keyed); err != nil {
			return fmt.Errorf("parse record: %w", err)
		}
		if err := b.Put(int64Key(keyed.Key), append([]byte(nil), line...)); err != nil {
			return fmt.Errorf("write record %d: %w", keyed.Key, err)
		}
	}
	return scanner.Err()
}

// int64Key encodes id as a big-endian byte slice suitable for use as a
// bbolt key.
func int64Key(id int64) []byte {
	key := make([]byte, 8)
	binary.BigEndian.PutUint64(key, uint64(id))
	return key
}

func latestZipInfo(ctx context.Context, indexURL, userAgent string) (etag, url string, err error) {
	body, err := fetchBytes(ctx, indexURL, userAgent)
	if err != nil {
		return "", "", fmt.Errorf("fetch index: %w", err)
	}
	var idx sdeIndex
	if err := json.Unmarshal(body, &idx); err != nil {
		return "", "", fmt.Errorf("parse index: %w", err)
	}
	for _, f := range idx.Files {
		if f.Name == sdeZipName {
			return f.Etag, f.URL, nil
		}
	}
	return "", "", fmt.Errorf("%s not found in sde index", sdeZipName)
}

func fetchBytes(ctx context.Context, url, userAgent string) ([]byte, error) {
	req, err := http.NewRequestWithContext(ctx, http.MethodGet, url, nil)
	if err != nil {
		return nil, fmt.Errorf("build request: %w", err)
	}
	req.Header.Set("User-Agent", userAgent)

	resp, err := http.DefaultClient.Do(req)
	if err != nil {
		return nil, fmt.Errorf("execute request: %w", err)
	}
	defer resp.Body.Close()
	if resp.StatusCode != http.StatusOK {
		return nil, fmt.Errorf("unexpected status %d from %s", resp.StatusCode, url)
	}
	return io.ReadAll(resp.Body)
}
```

```go
// Category is a minimal decode of an SDE category record -- the top level
// of EVE's item hierarchy (e.g. "Ship", "Module").
type Category struct {
	ID        int64
	Name      map[string]string
	Published bool

// Category looks up a category by ID. ok is false if the ID isn't in the
// cache -- either because the store has never synced successfully, or
// because no such category exists.
func (s *Store) Category(ctx context.Context, userAgent string, id int64) (Category, bool, error) {
	_ = s.ensureSynced(ctx, userAgent) // best-effort; see ensureSynced's doc comment

	var (
		out Category
		ok  bool
	)
	err := s.db.View(func(tx *bbolt.Tx) error {
		b := tx.Bucket([]byte("categories"))
		if b == nil {
			return nil
		}
		v := b.Get(int64Key(id))
		if v == nil {
			return nil
		}
		var raw struct {
			Key       int64             `json:"_key"`
			Name      map[string]string `json:"name"`
			Published bool              `json:"published"`
		}
		if err := json.Unmarshal(v, &raw); err != nil {
			return fmt.Errorf("parse cached category %d: %w", id, err)
		}
		out, ok = Category{ID: raw.Key, Name: raw.Name, Published: raw.Published}, true
		return nil
	})
	return out, ok, err
}
```
